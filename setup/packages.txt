# The Fedora packages of Frametop's two containers, in one list: the build image
# (pack/Containerfile: CI builds the programs for the SteamOS host there and runs the test
# gate) and the "dev" container on the Frame (setup/dev-container.sh: Remote Access's krdp,
# FreeRDP and TigerVNC). Both start from the same digest-pinned fedora-toolbox:44. Each line
# is "WHERE: packages  # why", WHERE being both, dev (the dev container only) or image (the
# image only). The libraries the programs link are ones the SteamOS host ships too: the image
# builds against Fedora's, and on the Frame they load the host's (pack/host-and-flatpak.md).
# pack/test/pins-test.py reads it as they do.

# toolchains (meson builds ft-screens' static wlroots; dev: also clang and Rust, for experiments)
both: gcc gcc-c++ cmake ninja-build make meson pkgconf-pkg-config git
dev: clang clang-devel cargo rust
dev: ShellCheck  # test-bash checks the installers' shared helper (scripts/install-lib.sh) with it
image: curl unzip just
# libraries for Frametop's native pieces (wlroots-devel: the C tests in the justfile; ft-screens
# builds its own static copy, screens/build.sh, since the host has none)
image: pipewire-devel libxkbcommon-devel libinput-devel systemd-devel dbus-devel libdrm-devel
image: mesa-libgbm-devel wayland-devel wayland-protocols-devel vulkan-loader-devel vulkan-headers
image: plasma-wayland-protocols wlroots-devel pixman-devel libffi-devel expat-devel libxml2-devel
# ft_pointer SteamVR driver and the testbench's preloads: static C++ runtime, hidden
# (vrserver has its own); ft-camd is linked statically, for its file capabilities
image: libstdc++-static glibc-static
# hand tracking: ft-hands reads the calibration with jsoncpp
image: jsoncpp-devel
# the hand recorder's export compresses recordings with zstd (its app has it in the Flatpak)
image: zstd
image: google-noto-sans-vf-fonts  # the visible reset countdown uses the installed Noto Sans font
# Frametop remote desktop (VNC bridge through krdp): what the dev container is for
dev: krdp freerdp tigervnc-x11-server xrandr
# Frametop Voice (voice/): glslc for whisper.cpp's Vulkan backend (the SPIR-V headers are
# fetched and pinned by voice/build.sh, since the host has none)
image: glslc
# Frametop's keyboard (keyboard/, TaiKeid's Full Keyboard): Cairo and Pango draw it, SDL2 for its
# preview, and its input methods: Hangul, Pinyin (pyzy), Japanese (Anthy, loaded at run time).
image: cairo-devel pango-devel json-c-devel sdl2-compat-devel libhangul-devel pyzy-devel anthy-unicode-devel
dev: libX11-devel  # its test probes (FULL_KEYBOARD_TESTS=1 keyboard/build.sh)
# remote displays (stream/): moonlight-common-c and moonlight-embedded's libgamestream,
# and the host's sound (Opus, played through PipeWire's PulseAudio server)
image: openssl-devel libcurl-devel libuuid-devel opus-devel pulseaudio-libs-devel
# diagnostics and remote UI testing
dev: wayland-utils xorg-x11-server-Xvfb ImageMagick xdotool
